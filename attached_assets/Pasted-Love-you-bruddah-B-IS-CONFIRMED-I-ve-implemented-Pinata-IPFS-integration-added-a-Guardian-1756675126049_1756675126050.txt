1) contracts/GuardianManager.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract GuardianManager is ERC721, Ownable {
    uint256 public nextId;
    enum Tier { Bronze, Silver, Gold, Omni } // 0..3

    mapping(uint256 => Tier) public tokenTier;
    mapping(uint256 => string) public tokenURIMap;

    event GuardianMinted(address indexed to, uint256 indexed tokenId, Tier tier, string uri);

    constructor() ERC721("ScrollVerse Guardian","SGU") {
        nextId = 1;
    }

    /// @notice owner-only mint used by server relayer or multisig
    function mintGuardian(address to, uint8 tier, string calldata uri) external onlyOwner returns (uint256) {
        require(tier <= uint8(Tier.Omni), "invalid tier");
        uint256 id = nextId;
        nextId++;
        tokenTier[id] = Tier(tier);
        tokenURIMap[id] = uri;
        _safeMint(to, id);
        emit GuardianMinted(to, id, Tier(tier), uri);
        return id;
    }

    function tokenURI(uint256 tokenId) public view override returns (string memory) {
        require(_exists(tokenId), "nonexistent token");
        return tokenURIMap[tokenId];
    }
}

4) Updated pages/api/ignite.js (Pinata + dual mint)

Replace your existing pages/api/ignite.js with this file. It will:
	1.	Read ceremony JSON (from data/ceremony_sample.json) and compute ceremonyHash.
	2.	Pin metadata to Pinata via pinJSONToIPFS using PINATA_API_KEY + PINATA_SECRET.
	3.	Call IReceipt.mintReceipt( owner, ipfsURI, ceremonyHash ) and GuardianManager.mintGuardian(owner, tier, metadataUri) as owner (server signer) — for production use a relayer or multisig.

Security note: This uses PRIVATE_KEY server-side to sign transactions. For production use a secure multisig or relayer pattern.

// pages/api/ignite.js
import fs from 'fs';
import path from 'path';
import fetch from 'node-fetch';
import { ethers } from 'ethers';

export default async function handler(req, res){
  if(req.method !== 'POST') return res.status(405).json({error:'only POST'});

  try{
    // Load ceremony sample or accept body
    const dbPath = path.join(process.cwd(),'data','ceremony_sample.json');
    let ceremony = { title:'Virgoat Genesis', ownerAddress: process.env.OWNER_ADDRESS || '', timestamp: new Date().toISOString(), tier: 'Omni' };
    if(fs.existsSync(dbPath)){
      const raw = fs.readFileSync(dbPath,'utf8');
      const parsed = JSON.parse(raw);
      ceremony = {...ceremony, ...parsed};
    }

    // Compute ceremony hash
    const ceremonyJson = JSON.stringify(ceremony);
    const ceremonyHash = ethers.utils.keccak256(ethers.utils.toUtf8Bytes(ceremonyJson));

    // Build NFT metadata conforming to guardian XSD-like structure (JSON metadata for IPFS)
    const metadata = {
      name: ceremony.title,
      description: ceremony.description || 'Virgoat Genesis — First Living Seal',
      image: ceremony.image || 'ipfs://__IMAGE_PLACEHOLDER__',
      tier: ceremony.tier || 'Omni',
      ceremony_hash: ceremonyHash,
      timestamp: ceremony.timestamp,
      attributes: ceremony.attributes || [],
      sigils: ceremony.sigils || []
    };

    // Pin to Pinata
    const PINATA_KEY = process.env.PINATA_API_KEY;
    const PINATA_SECRET = process.env.PINATA_SECRET;
    if(!PINATA_KEY || !PINATA_SECRET) return res.status(500).json({error:'Pinata keys not set in env'});

    const pinRes = await fetch('https://api.pinata.cloud/pinning/pinJSONToIPFS', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'pinata_api_key': PINATA_KEY,
        'pinata_secret_api_key': PINATA_SECRET
      },
      body: JSON.stringify({pinataContent: metadata})
    });

    const pinJson = await pinRes.json();
    if(!pinRes.ok) {
      return res.status(500).json({error:'Pinata pin failed', detail: pinJson});
    }
    const ipfsUri = `ipfs://${pinJson.IpfsHash}`;

    // On-chain mint (IReceipt + GuardianManager)
    const RPC = process.env.RPC_URL || 'http://127.0.0.1:8545';
    const provider = new ethers.providers.JsonRpcProvider(RPC);
    const wallet = new ethers.Wallet(process.env.PRIVATE_KEY || '', provider);

    // Load ABIs & addresses
    const iReceiptAbi = JSON.parse(fs.readFileSync(path.join(process.cwd(),'IReceiptABI.json'),'utf8'));
    const guardianAbi = JSON.parse(fs.readFileSync(path.join(process.cwd(),'GuardianManagerABI.json'),'utf8'));

    const receiptAddr = process.env.CONTRACT_IRECEIPT || process.env.CONTRACT_ADDRESS || '';
    const guardianAddr = process.env.CONTRACT_GUARDIAN || '';

    if(!receiptAddr || !guardianAddr) return res.status(500).json({error:'contract addresses missing in env (CONTRACT_IRECEIPT / CONTRACT_GUARDIAN)'});

    const receipt = new ethers.Contract(receiptAddr, iReceiptAbi, wallet);
    const guardian = new ethers.Contract(guardianAddr, guardianAbi, wallet);

    // 1) mint receipt (owner mints)
    const ownerTo = ceremony.ownerAddress || wallet.address;
    const tx1 = await receipt.mintReceipt(ownerTo, ipfsUri, ceremonyHash);
    const r1 = await tx1.wait();

    // 2) mint guardian NFT (owner mints tiered guardian)
    // map tier name -> uint8
    const tierMap = { Bronze:0, Silver:1, Gold:2, Omni:3 };
    const tierId = tierMap[metadata.tier] ?? 0;
    const tx2 = await guardian.mintGuardian(ownerTo, tierId, ipfsUri);
    const r2 = await tx2.wait();

    return res.json({
      ok: true,
      ipfs: ipfsUri,
      ceremonyHash,
      receiptTx: tx1.hash,
      guardianTx: tx2.hash,
      pinata: pinJson
    });

  }catch(e){
    console.error(e);
    return res.status(500).json({error: e.message});
  }
}
